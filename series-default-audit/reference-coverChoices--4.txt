
    };
    const priorityJobs = priority.map(load).filter(Boolean);
    homeHeroReady = !priorityJobs.length;
    if (priorityJobs.length) {
      Promise.allSettled(priorityJobs).then(() => {
        homeHeroReady = true;
        observeDeferred();
        releaseLazyBackgrounds();
      });
    } else {
      observeDeferred();
      releaseLazyBackgrounds();
    }
  }
